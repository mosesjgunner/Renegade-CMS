import { readFileSync, writeFileSync } from 'node:fs'
import { jwtVerify } from 'jose'
const env=Object.fromEntries(readFileSync('.env.compose','utf8').trim().split(/\r?\n/).map(l=>{const n=l.indexOf('=');return [l.slice(0,n),l.slice(n+1)]}))
const state=JSON.parse(readFileSync('scratch/image-owner.json','utf8'))
const token=state.cookies.find(c=>c.name==='renegade-passkey').value
await jwtVerify(token,new TextEncoder().encode(env.PAYLOAD_SECRET))
let buildDefaultAccepted=false
try { await jwtVerify(token,new TextEncoder().encode('next-build-value-not-used-at-runtime-0123456789'));buildDefaultAccepted=true } catch {}
if(buildDefaultAccepted) throw new Error('Build-only signing value accepted in production')
writeFileSync('scratch/sol-evidence/runtime-signing-secret.json',JSON.stringify({legitimateBrowserSessionVerifiedWithConfiguredRuntimeSecret:true,buildDefaultAccepted:false,secretDisclosed:false},null,2))
console.log('Runtime-issued session uses the configured deployment secret.')
