import fs from 'node:fs'
import path from 'node:path'
import { createRequire } from 'node:module'
const packages = [], unversionedManifests = [], traces = [], suspectFiles = []
function walk(dir) {
  for (const entry of fs.readdirSync(dir, { withFileTypes: true })) {
    const file = path.join(dir, entry.name)
    if (entry.isSymbolicLink()) continue
    if (entry.isDirectory()) walk(file)
    else if (entry.name === 'package.json') {
      try { const pkg = JSON.parse(fs.readFileSync(file)); if(pkg.name && pkg.version) packages.push({ path:file, name:pkg.name, version:pkg.version }); else unversionedManifests.push({path:file,name:pkg.name??null,version:pkg.version??null}) } catch {}
    } else if (file.endsWith('.nft.json')) {
      const trace = JSON.parse(fs.readFileSync(file)); traces.push({path:file, files:trace.files?.length, missing:(trace.files??[]).filter(f=>!fs.existsSync(path.resolve(path.dirname(file),f)))})
    } else if (/^\.env($|\.)|^id_rsa$|^.*\.(key|pem)$/.test(entry.name)) suspectFiles.push(file)
  }
}
