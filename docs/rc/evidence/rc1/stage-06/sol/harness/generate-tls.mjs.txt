import { webcrypto, randomBytes } from 'node:crypto'
import { mkdirSync, writeFileSync } from 'node:fs'
import * as x509 from '@peculiar/x509'

x509.cryptoProvider.set(webcrypto)
const algorithm = {
  name: 'RSASSA-PKCS1-v1_5',
  hash: 'SHA-256',
  publicExponent: new Uint8Array([1, 0, 1]),
  modulusLength: 2048,
}
const keys = await webcrypto.subtle.generateKey(algorithm, true, ['sign', 'verify'])
const certificate = await x509.X509CertificateGenerator.createSelfSigned({
  serialNumber: randomBytes(16).toString('hex'),
  name: 'CN=dispatch.rc02.test',
  notBefore: new Date(Date.now() - 60_000),
  notAfter: new Date(Date.now() + 365 * 24 * 60 * 60 * 1000),
  signingAlgorithm: algorithm,
  keys,
  extensions: [
    new x509.BasicConstraintsExtension(true, 0, true),
    new x509.SubjectAlternativeNameExtension([{ type: 'dns', value: 'dispatch.rc02.test' }]),
  ],
})
mkdirSync('scratch/tls-final', { recursive: true })
const privateKey = Buffer.from(await webcrypto.subtle.exportKey('pkcs8', keys.privateKey))
writeFileSync(
  'scratch/tls-final/key.pem',
  `-----BEGIN PRIVATE KEY-----\n${privateKey
    .toString('base64')
    .match(/.{1,64}/g)
    .join('\n')}\n-----END PRIVATE KEY-----\n`,
  { mode: 0o600 },
)
writeFileSync('scratch/tls-final/cert.pem', certificate.toString('pem'))
console.log('Certificate generated successfully')
