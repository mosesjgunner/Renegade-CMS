from pathlib import Path
import subprocess, os, json, secrets, time
out = Path('scratch/review-checks')
sha = subprocess.check_output(['git','rev-parse','HEAD'], text=True).strip()
password = secrets.token_hex(24)
name = 'sol-gemini-review-gates-20261016'
assert name not in subprocess.check_output(['docker','ps','-a','--format','{{.Names}}'],text=True).splitlines()
subprocess.run(['docker','run','-d','--name',name,'--label','purpose=disposable-sol-gemini-review','-e','POSTGRES_USER=renegade','-e','POSTGRES_PASSWORD='+password,'-p','127.0.0.1::5432','postgres:17.6-alpine'],check=True,stdout=subprocess.DEVNULL)
for attempt in range(30):
    if subprocess.run(['docker','exec',name,'pg_isready','-U','renegade'],stdout=subprocess.DEVNULL,stderr=subprocess.DEVNULL).returncode == 0: break
    time.sleep(1)
else: raise RuntimeError('Disposable PostgreSQL did not become ready')
port = subprocess.check_output(['docker','port',name,'5432'],text=True).strip().split(':')[-1]
env = os.environ.copy()
env.update(PAYLOAD_SECRET=secrets.token_hex(48), APP_URL='http://localhost:3000', LOCAL_E2E_TEST_MODE='false', RENEGADE_MODULES='all', RENEGADE_ALLOW_UNSAFE_COLLECTION_COUNT='true', BUILD_SHA=sha)
for database in ['review_release_acceptance','review_upgrade_acceptance','commerce_release_acceptance']:
    subprocess.run(['docker','exec',name,'createdb','-U','renegade',database],check=True)
base = f'postgresql://renegade:{password}@127.0.0.1:{port}/'
env.update(DATABASE_URL=base+'review_release_acceptance', INTEGRATION_DATABASE_URL=base+'review_release_acceptance', UPGRADE_MIGRATION_DATABASE_URL=base+'review_upgrade_acceptance')
receipts=[]
gates=[('fresh',['npm.cmd','run','test:migrations:fresh']),('upgrade-fixture',['npm.cmd','run','test:migrations:upgrade']),('commerce',['npm.cmd','run','test:migrations:commerce']),('integration',['npm.cmd','run','test:integration']),('build',['npm.cmd','run','build']),('bundles',['npm.cmd','run','verify:presentation-bundles']),('image',['docker','build','--target','runner','--build-arg','BUILD_SHA='+sha,'-t','renegade-cms:sol-review-'+sha[:8],'.'])]
for label,args in gates:
    gateEnv=env.copy()
    if label=='commerce': gateEnv['DATABASE_URL']=base+'commerce_release_acceptance'
    start=time.time()
    with (out/(label+'.log')).open('wb') as log:
        result=subprocess.run(args,env=gateEnv,stdout=log,stderr=subprocess.STDOUT)
    text=(out/(label+'.log')).read_text(errors='replace')
    for secret in [password,env['PAYLOAD_SECRET']]: text=text.replace(secret,'[REDACTED]')
    (out/(label+'.log')).write_text(text,encoding='utf-8')
    receipts.append(dict(gate=label,command=args,sourceSha=sha,exitCode=result.returncode,durationSeconds=round(time.time()-start,2),disposableDatabaseContainer=name,disposablePort=port))
    (out/'runtime-gates.json').write_text(json.dumps(receipts,indent=2)+'\n')
    print(label,'exit',result.returncode,flush=True)
    if result.returncode: raise SystemExit(result.returncode)
