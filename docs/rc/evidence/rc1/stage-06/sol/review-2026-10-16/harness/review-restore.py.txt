def run(label,args,expected=0):
    started=time.time()
    with (out/(label+'.log')).open('wb') as log:
        p=subprocess.run(args,stdout=log,stderr=subprocess.STDOUT)
    text=(out/(label+'.log')).read_text(errors='replace')
    for secret in secretsUsed: text=text.replace(secret,'[REDACTED]')
    (out/(label+'.log')).write_text(text,encoding='utf-8')
    receipts.append(dict(gate=label,command=args,sourceSha=sha,exitCode=p.returncode,expectedExit=expected,durationSeconds=round(time.time()-started,2)))
    (out/'restore-gates.json').write_text(json.dumps(receipts,indent=2)+'\n')
