const request = (path: string, cookie = '', method = 'GET', body?: any, headers: any = {}) =>
  fetch(base + path, {
    method,
    headers: {
      origin: base,
      ...(cookie ? { cookie } : {}),
      ...(body && !(body instanceof FormData) ? { 'content-type': 'application/json' } : {}),
      ...headers,
    },
    body:
      body instanceof FormData
        ? body
        : body === undefined
          ? undefined
          : typeof body === 'string'
            ? body
            : JSON.stringify(body),
    signal: AbortSignal.timeout(180000),
  })
const json = async (
  path: string,
  cookie = '',
  method = 'GET',
  body?: any,
  status = 200,
  headers: any = {},
) => {
  const res = await request(path, cookie, method, body, headers)
  const text = await res.text()
  assert.equal(res.status, status, `${method} ${path}: ${text.slice(0, 1500)}`)
  return status >= 500 ? { raw: text } : JSON.parse(text)
}

async function signed(
  id: string,
  type: string,
  object: any,
  expected = 202,
  created = nextEventTime(),
) {
  const raw = JSON.stringify({
    id,
    type,
    created,
    livemode: false,
    data: { object },
  })
  const t = Math.floor(Date.now() / 1000),
    sig = createHmac('sha256', process.env.COMMERCE_STRIPE_TEST_WEBHOOK_SECRET!)
      .update(`${t}.${raw}`)
      .digest('hex')
  return json('/api/commerce/webhooks/stripe-test', '', 'POST', raw, expected, {
    'stripe-signature': `t=${t},v1=${sig}`,
  })
}

        './tests/rc1-12b/stripe-preload.mjs',
        '--import',
        'tsx',
        'src/scripts/run-jobs-worker.ts',
      ],
      {
        env: {
          ...process.env,
          WORKER_PROFILE: 'commerce-only',
          WORKER_HEARTBEAT_FILE: heartbeat,
          WORKER_POLL_INTERVAL_MS: '1000',
        },
        windowsHide: true,
        stdio: ['ignore', 'pipe', 'pipe'],
      },
    )
    const log = createWriteStream(`${out}/${label}.log`)
    worker.stdout!.pipe(log, { end: false })
    worker.stderr!.pipe(log, { end: false })
    try {
      const deadline = Date.now() + 120000
      while (Date.now() < deadline) {
        if (interrupt && providerBlocked) return
        try {
          const value = JSON.parse(await readFile(heartbeat, 'utf8'))
          if (
            !interrupt &&
            value.pid === worker.pid &&
            (await row('payload-jobs', pendingBillingJob)).completedAt
          )
            return
        } catch {}
        if (worker.exitCode != null) throw new Error(`Worker exited ${worker.exitCode}`)
        await new Promise((resolve) => setTimeout(resolve, 250))
      }
      throw new Error(`Worker timed out: ${label}`)
    } finally {
      if (worker.pid)
        await new Promise((resolve) =>
          spawn('taskkill', ['/pid', String(worker.pid), '/t', '/f'], {
            windowsHide: true,
            stdio: 'ignore',
          }).once('exit', resolve),
        )
      log.end()
    }
  }
  const queue = async () => {
    const queued = await json(
      '/api/admin/commerce/reconcile',
      operator,
      'POST',
      { siteId: state.alpha, kind: 'subscriptions' },
      202,
    )

  }
  if (!restart) {

    await ctx.addCookies([
      { name: 'renegade-member', value: sessions.member.sessionToken, url: base },

      await page.getByRole('heading', { name: 'Isolated billing portal' }).waitFor()
      await page.getByRole('button', { name: action, exact: true }).click()
      await page.getByRole('button', { name: 'Manage billing for', exact: false }).waitFor()
    }
    const invoice = (reference: string, paid: boolean) => ({
      object: 'invoice',
      id: reference,
      subscription: harnessRecord.id,
      customer: harnessRecord.customer,
      status: paid ? 'paid' : 'open',
      amount_paid: paid ? 1200 : 0,
      amount_due: 1200,
      currency: 'usd',
    })
    harnessRecord.current_period_end += 2592000
    await signed('evt_renewal12b', 'invoice.paid', invoice('in_renewal12b', true))
    check(
      'signed renewal extends persisted period and exactly two stable grants',
      (await row('subscriptions', state.subscription)).currentPeriodEnd ===
        new Date(harnessRecord.current_period_end * 1000).toISOString() &&
        (await snapshot()).grants.length === 2,
    )
    await billing()
    check(
      'member billing shows renewal invoice paid and next billing',
      (await page.locator('#billing').innerText()).includes('Invoice in_renewal12b: paid') &&
        (await listing()).nextBillingAt,
    )
    const renewalVersion = (await row('subscriptions', state.subscription)).version
    await Promise.all([
      signed('evt_renewal12b', 'invoice.paid', invoice('in_renewal12b', true), 200),
      signed('evt_renewal12b', 'invoice.paid', invoice('in_renewal12b', true), 200),
    ])
    check(
      'concurrent renewal replay leaves version and invoice count unchanged',
      (await row('subscriptions', state.subscription)).version === renewalVersion &&
        (await listing()).invoices.filter((i: any) => i.reference === 'in_renewal12b').length === 1,
    )
    providerUnavailable = true
    const beforeUnknown = await snapshot()
    await signed('evt_unknown12b', 'invoice.paid', invoice('in_unknown12b', true), 500)
    check(
      'provider error rolls back signed event and cannot manufacture renewal success',
      JSON.stringify(await snapshot()) === JSON.stringify(beforeUnknown),
    )
    await queue()
    await workerCycle('unknown-worker')
    check(
      'worker exposes unknown provider state while retaining bounded last-known access',
      (await listing()).reconciliationUnknown === true && (await canRead()),
    )
    providerUnavailable = false
    await signed('evt_unknown12b', 'invoice.paid', invoice('in_unknown12b', true))
    check(
      'exact event succeeds once after provider recovery',
      !(await listing()).reconciliationUnknown &&
        (await listing()).invoices.some(
          (i: any) => i.reference === 'in_unknown12b' && i.state === 'paid',
        ),
    )
    harnessRecord.status = 'past_due'
    await signed('evt_failure12b', 'invoice.payment_failed', invoice('in_failed12b', false))
    const boundedFailure = await snapshot()
    check(
      'failed invoice atomically caps entitlements at grace before any worker runs',
      boundedFailure.subscription.state === 'grace' &&
        boundedFailure.grants.every(
          (grant: any) => grant.endsAt === boundedFailure.subscription.graceEnd,
        ),
    )
    const beforeInterrupt = await snapshot()
    await queue()
    blockProvider = true
    await workerCycle('interrupted-worker', true)
    blockProvider = false
    check(
      'killed worker leaves subscription ledger and entitlements atomic and unchanged',
      JSON.stringify(await snapshot()) === JSON.stringify(beforeInterrupt),
    )
    await new Promise((resolve) => setTimeout(resolve, 1200))
    check(
      'canonical recovery reopens abandoned billing job lease',
      (await recoverStaleCommerceJobs(payload, 1000)) >= 1,
    )
    await workerCycle('recovered-worker')
    const grace = await row('subscriptions', state.subscription)
    check(
      'fresh worker applies dunning grace and one in-app notice',
      grace.state === 'grace' &&
        (await canRead()) &&
        (await find('activity-events', { type: { equals: 'billing.payment_action_required' } }))
          .length === 1,
    )
    await billing()
    check(
      'member observes failed invoice and contractual grace deadline',

    check(
      'worker retry preserves grace deadline and deduplicates dunning notices',

        (await find('activity-events', { type: { equals: 'billing.payment_action_required' } }))
          .length === 1,
    )
    console.log('Waiting for the isolated contractual grace deadline', grace.graceEnd)
    while (Date.now() <= Date.parse(grace.graceEnd))
      await new Promise((resolve) => setTimeout(resolve, 500))
    check(
      'canonical access expires at the saved grace deadline with worker stopped',
      !(await canRead()),
    )

    await workerCycle('grace-expiry-worker')
    check(
      'expired grace revokes canonical access while invoice remains failed',
      (await row('subscriptions', state.subscription)).state === 'expired' &&
        !(await canRead()) &&
        (await listing()).invoices.some(
          (i: any) => i.reference === 'in_failed12b' && i.state === 'payment_failed',
        ),
    )
    await queue()
    await workerCycle('expired-retry-worker')
    check(
      'expired retry never extends grace or duplicates expiry notice',

        (await find('activity-events', { type: { equals: 'billing.access_expired' } })).length ===

    check(
      'successful payment recovery restores stable grants and paid invoice',

        (await listing()).invoices.some(
          (i: any) => i.reference === 'in_failed12b' && i.state === 'paid',

      (await row('subscriptions', state.subscription)).state === 'cancel_at_period_end' &&

        (await canRead()),

      'invoice arriving after cancel request cannot unschedule provider cancellation',

    check(
      'customer can withdraw scheduled cancellation through hosted billing',
      (await row('subscriptions', state.subscription)).state === 'active' &&
        (await listing()).nextBillingAt,
    )
    await billing()
    await page.getByLabel('Preview a plan change').selectOption(state.year)
    await page.getByText('Provider change estimate:', { exact: false }).waitFor()
    check(
      'customer sees actual adapter plan change preview',
      (await page.locator('#billing').innerText()).includes('108.00'),

    await portal('change')
    check(
      'customer plan change persists mapped immutable annual agreement',
      (await row('subscriptions', state.subscription)).planRevision === state.year &&
        (await listing()).plan.name === 'Journal annual 12B' &&
        (await canRead()),

      '/api/admin/commerce/reconcile',

  const idle = new Set(pool._idle.map((entry: any) => entry.client))

      client.release(true)
    }
  await pool.end()
  await payload.destroy()
}
