from pathlib import Path
import json, hashlib
base = Path('C:/Projects/RENEGADE CMS/sol-rc04-20261014/scratch/sol-stage04')
fixture = json.loads(Path('scratch/records/source-business.json').read_text())
dest = Path('scratch/resume-business-specs'); dest.mkdir(exist_ok=True)
Path('scratch/resume-business-results').mkdir(exist_ok=True)
manifest = []
for name in ['acceptance.spec.ts', 'commerce.spec.ts']:
    source = base/name
    text = source.read_text()
    text = text.replace('scratch/sol-owner-authenticator.json', 'scratch/owner-device.json')
    text = text.replace('editor@renegadeparty.test', 'owner@rc06.example.test')
    text = text.replace('where[slug][equals]=renegadeparty', 'where[id][equals]='+fixture['siteId'])
    text = text.replace('scratch/sol-stage04-results', 'scratch/resume-business-results')
    # Genuine storage state is obtained by the first test's ordinary WebAuthn login.
    text = text.replace('scratch/rc04-operator.json', 'scratch/resume-operator.json')
    target = dest/name; target.write_text(text)
    manifest.append({'original':str(source),'originalSha256':hashlib.sha256(source.read_bytes()).hexdigest(),'adapted':str(target),'adaptedSha256':hashlib.sha256(target.read_bytes()).hexdigest()})
config = Path('scratch/playwright.config.ts').read_text().replace("testDir:'./specs'", "testDir:'./resume-business-specs'")
Path('scratch/resume-business.config.ts').write_text(config)
Path('scratch/records/resume-business-harness.json').write_text(json.dumps({'testAuthentication':False,'sessionCreation':'ordinary owner WebAuthn and member MIME links; no minted sessions','adaptations':'existing disposable site ID, operator email/device file, private output paths and dedicated spec directory; assertions unchanged','specs':manifest},indent=2))
print('Prepared source-hashed ordinary business population harness; no product state mutated')
