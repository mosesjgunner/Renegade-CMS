import fs from 'node:fs'
const file='docs/rc/CAPABILITY_LEDGER.md'
const text=fs.readFileSync(file,'utf8')
const start=text.indexOf('## Platform/Operations')
const end=text.indexOf('## Internal Payload registrations')
const groups=['Platform/Operations','Identity','Publishing/Editorial','Presentation/Themes/Puck','Media/DAM','Discovery/Search/SEO/Quality','Workflow/Releases','Social/Distribution/Network','Audience/Email/Telecom','Community','Analytics/Experiences','Commerce/Fulfillment','Integrations/Providers']
const data=Object.fromEntries(groups.map(g=>[g,[]]));let current=''
let header=''
for(const l of text.slice(start,end).split('\n')){
 if(l.startsWith('## '))current=l.slice(3).trim()
 if(!l.startsWith('|'))continue
 const r=l.split('|').slice(1,-1).map(s=>s.trim())
 if(r[0]==='Capability ID'){if(!header)header=l+'\n';continue}
 if(r[0].startsWith('--')){if(header.split('\n').length===2)header+=l+'\n';continue}
 let group=current;const slug=r[0].replace(/^(collections|globals|tasks):/,'')
 if(slug==='page-layouts')group='Presentation/Themes/Puck'
 if(/^(media-|albums$)/.test(slug))group='Media/DAM'
 if(/^(taxonomy-redirects|public-redirects)$/.test(slug))group='Discovery/Search/SEO/Quality'
 if(/^(forum|discussion|relationships$|work-conversations|work-messages)/.test(slug))group='Community'
 if(/^(team-|editorial-assignments|editorial-discussions|editorial-comments|realtime-|scheduled-publish-jobs|revision-records|calendar-entries)/.test(slug))group='Workflow/Releases'
 data[group].push(l)
}
const out=text.slice(0,start)+groups.map(g=>'## '+g+'\n\n'+header+data[g].join('\n')+'\n\n').join('')+text.slice(end)
fs.writeFileSync(file,out)
const sourceCandidates=['.env.example','README.md','docker-compose.yml','compose.yml','compose.yaml'].filter(f=>fs.existsSync(f))
const matches=[]
for(const f of sourceCandidates)fs.readFileSync(f,'utf8').split(/\r?\n/).forEach((line,i)=>{if(/(?:PASSWORD|SECRET|TOKEN|KEY)\s*=|changeme|example.*credential/i.test(line))matches.push({file:f,line:i+1,classification:'Example/build/documentation candidate; no value copied, not proof of a production credential leak'})})
fs.writeFileSync('docs/rc/evidence/example-credential-locations.json',JSON.stringify({scope:'Example configuration and root documentation; real .env files excluded',matches},null,2)+'\n')
const hashes=JSON.parse(fs.readFileSync('docs/rc/evidence/historical-preservation-check.json','utf8'))
const coverage=JSON.parse(fs.readFileSync('docs/rc/evidence/coverage-review.json','utf8'))
if(hashes.some(h=>!h.originalCheckoutBytesPreserved)||coverage.coverageIssues.length)throw new Error('Inventory validation failed')
const rowIds=out.split('\n').filter(l=>l.startsWith('|')).map(l=>l.split('|')[1].trim()).filter(id=>/^(collections|globals|tasks):/.test(id))
if(rowIds.length!==240||new Set(rowIds).size!==240)throw new Error('Registration status uniqueness failed after regrouping')
console.log('Ledger regrouped by product area; 240 unique registration statuses preserved')
